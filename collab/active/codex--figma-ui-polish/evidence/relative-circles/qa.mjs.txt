async function settled(page){

 }

function geometry(s,label){
 const m=s.markers,unique=[...new Set(m.map(v=>v.count))].sort((a,b)=>a-b),overlap=[];
 for(let i=0;i<m.length;i++)for(let j=i+1;j<m.length;j++){const d=Math.hypot(m[i].cx-m[j].cx,m[i].cy-m[j].cy)-(m[i].diameter+m[j].diameter)/2;if(d<-.75)overlap.push({a:m[i].label,b:m[j].label,clearance:d});}
 check(m.length>0&&m.every(v=>v.inside&&Number.isFinite(v.diameter)&&v.diameter>0&&Math.abs(v.rect.height-v.diameter)<.2),label+' finite circles inside camera');
 check(overlap.length===0,label+' no circle-fill overlap',{overlap});check(m.every(v=>v.textWidth<=v.diameter-2&&v.textHeight<=v.diameter-2),label+' count text fits');
 for(const n of unique){const a=m.filter(v=>v.count===n);check(Math.max(...a.map(v=>v.diameter))-Math.min(...a.map(v=>v.diameter))<.5,label+' equal count equal diameter '+n);}
 for(let i=1;i<unique.length;i++)check(m.find(v=>v.count===unique[i]).diameter>m.find(v=>v.count===unique[i-1]).diameter+.5,label+' larger count larger diameter '+unique[i-1]+'→'+unique[i]);
}
