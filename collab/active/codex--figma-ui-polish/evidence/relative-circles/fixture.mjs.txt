function makeDataset(specs,seed){
 const items=[],mapItems=[];
 for(const [key,label,count,latitude,longitude,recent] of specs)for(let i=0;i<count;i++){
  const id=(seed++).toString(16).padStart(64,'0'),age=i<recent?(i+1)*hour:(50+i)*hour;
  const item={collection_id:'relative-circles-qa',article_id:id,image_status:'ready',image_url:'/images/'+id+'.jpg',article:{title:'검증용 기사 · '+label+' · '+(i+1)+'번째 보도',description:'지역 숫자 원과 선택시트 상태 확인을 위한 합성 기사입니다. 실제 사건 보도가 아닙니다. 안전 안내를 정리하는 검증용 문장입니다.',pubDate:new Date(now-age).toISOString(),originallink:'https://publisher.example.invalid/relative/'+id,link:''}};
  items.push(item);mapItems.push({...item,location_status:'resolved',location_reason:null,location:{latitude,longitude,label,precision:'region',evidence:{field:'title',text:label}}});
 }
 items.sort((a,b)=>Date.parse(b.article.pubDate)-Date.parse(a.article.pubDate));
 return {items,mapItems,groups:specs.map(([key,label,count,,,recent])=>({key,label,count,recent}))};
}

function reply(res,status,body){res.writeHead(status,{'Content-Type':'application/json; charset=utf-8','Cache-Control':'no-store','Access-Control-Allow-Origin':'*'});res.end(JSON.stringify(body));}
const server=createServer((req,res)=>{
 const u=new URL(req.url||'/','http://fixture.invalid'),p=u.pathname;
 if(req.method!=='GET'&&req.method!=='HEAD')return reply(res,405,{error:'QA_READ_ONLY'});
 if(p==='/__qa/health')return reply(res,200,{ok:true,synthetic:true,selected,total:data[selected].items.length,groups:data[selected].groups});
 if(p==='/__qa/dataset'){const name=u.searchParams.get('name');if(!Object.hasOwn(data,name))return reply(res,400,{error:'UNKNOWN_DATASET'});selected=name;return reply(res,200,{selected});}
 requests.push({method:req.method,path:p,query:u.search});
 if(p==='/news'){const q=(u.searchParams.get('keyword')||'').toLowerCase(),all=data[selected].items.filter(i=>!q||i.article.title.toLowerCase().includes(q)),offset=Number(u.searchParams.get('offset')||0),limit=Number(u.searchParams.get('limit')||20);return reply(res,200,{total:all.length,items:all.slice(offset,offset+limit)});}
 if(p==='/map/articles'){const all=data[selected].mapItems,offset=Number(u.searchParams.get('offset')||0),limit=Number(u.searchParams.get('limit')||100);return reply(res,200,{total:all.length,items:all.slice(offset,offset+limit)});}

 if(/^\/images\/[a-f0-9]{64}\.jpg$/.test(p)){res.writeHead(200,{'Content-Type':'image/svg+xml','Cache-Control':'no-store','Access-Control-Allow-Origin':'*'});return res.end(image);}
 return reply(res,404,{error:'QA_NOT_FOUND'});
});
