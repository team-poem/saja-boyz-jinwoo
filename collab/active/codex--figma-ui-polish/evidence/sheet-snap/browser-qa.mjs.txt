const contexts = new Set();
async function pageFor(options = {}) {
  const { manualShareFailure = false, ...rest } = options;
  const context = await browser.newContext({
    viewport: { width: 393, height: 852 }, reducedMotion: 'reduce',
    permissions: ['geolocation'], geolocation: { latitude: 37.56, longitude: 126.99, accuracy: 1000 },
    ...rest,
  });
  contexts.add(context);
  // No browser route interception or SDK stub is installed.
  if (manualShareFailure) await context.addInitScript(() => {
    Object.defineProperty(navigator, 'share', { configurable: true, value: async () => { throw new Error('Synthetic QA share unavailable'); } });
    Object.defineProperty(navigator, 'clipboard', { configurable: true, value: { writeText: async () => { throw new Error('Synthetic QA clipboard unavailable'); } } });
  });
  const page = await context.newPage();
  page.on('pageerror', error => report.pageErrors.push(error.message));
  page.on('request', request => {
    if (!['GET', 'HEAD', 'OPTIONS'].includes(request.method())) {
      const url = new URL(request.url());
      const entry = { method: request.method(), origin: url.origin, url: url.origin + url.pathname, path: url.pathname, queryRedacted: Boolean(url.search) };
      if ([new URL(origin).origin, new URL(api).origin].includes(url.origin)) report.observedWrites.push(entry);
      else report.externalWrites.push(entry);
    }
  });
  page.on('response', response => {
    if (new URL(response.url()).hostname === 'nrbe.map.naver.net' && response.status() === 200) report.tiles200++;
  });
  return page;
}
async function scenario(name, task) {
  const entry = { name, completed: false, started: Date.now() };
  report.scenarios.push(entry);
  try { await task(); entry.completed = true; }
  catch (error) { entry.error = String(error?.stack || error); check(false, name + ' completed', entry.error); }
  finally {
    for (const context of [...contexts]) { contexts.delete(context); await context.close().catch(() => {}); }
    entry.elapsedMs = Date.now() - entry.started;
  }
}

async function metrics(dialog) {
  return dialog.evaluate(panel => {
    const reading = [...panel.children].find(child => child.getAttribute('tabindex') === '0');
    if (!reading) throw new Error('Reading scroller missing');
    const p = panel.getBoundingClientRect(), r = reading.getBoundingClientRect();
    const last = reading.lastElementChild?.getBoundingClientRect();
    const previous = reading.lastElementChild?.previousElementSibling?.getBoundingClientRect();
    const readingGap = parseFloat(getComputedStyle(reading).rowGap) || 0;
    return {
      panel: p.toJSON(), reading: r.toJSON(),
      expanded: panel.getAttribute('data-expanded') === 'true',
      viewport: { width: innerWidth, height: innerHeight },
      scrollTop: reading.scrollTop, clientHeight: reading.clientHeight, scrollHeight: reading.scrollHeight,
      clientWidth: reading.clientWidth, scrollWidth: reading.scrollWidth,
      endGap: last ? r.bottom - last.bottom : null,
      footerGap: last && previous ? last.top - previous.bottom : null, readingGap,
      paddingBottom: parseFloat(getComputedStyle(panel).paddingBottom),
      bodyScrollY: scrollY,
    };
  });
}
async function settled(dialog, label) {
  await dialog.page().evaluate(() => document.fonts.ready.then(() => undefined));
  let previous = '', consecutive = 0, value;
  const started = Date.now();
  for (let attempt = 0; attempt < 60; attempt++) {
    value = await metrics(dialog);
    const signature = JSON.stringify([value.panel.height, value.panel.width, value.clientHeight, value.scrollHeight, value.scrollTop].map(number => Math.round(number * 10)));
    consecutive = signature === previous ? consecutive + 1 : 0;
    if (value.panel.height > 0 && Date.now() - started >= 350 && consecutive >= 3) {
      if (label) report.measurements.push({ label, ...value });
      return value;
    }
    previous = signature;
    await new Promise(resolve => setTimeout(resolve, 100));
  }
  throw new Error('Sheet geometry did not settle: ' + label);
}
function geometryChecks(value, label) {
  const cap = value.expanded ? value.viewport.height - 49 : value.viewport.height * 0.5352;
  check(value.panel.height <= cap + 1.5, label + ' respects the current height ceiling', { actual: value.panel.height, cap, expanded: value.expanded });
  check(value.panel.top >= -1 && value.panel.bottom <= value.viewport.height + 1, label + ' stays inside the viewport', value.panel);
  check(value.clientHeight > 0, label + ' leaves an actual reading area');
  check(value.scrollWidth <= value.clientWidth + 1, label + ' has no horizontal reading overflow');
  if (value.expanded) check(Math.abs(value.panel.height - cap) <= 1.5, label + ' manual expansion uses the explicit near-full height', { actual: value.panel.height, cap });
  if (!value.expanded && value.scrollHeight <= value.clientHeight + 1) check(value.endGap !== null && Math.abs(value.endGap) <= 3, label + ' default state has no unused space below its last flow content', { endGap: value.endGap });
}
async function setExpanded(dialog, expanded) {
  const handle = dialog.locator('button[aria-expanded]').first();
  if ((await handle.getAttribute('aria-expanded')) !== String(expanded)) {
    const page = dialog.page(), box = await handle.boundingBox();
    must(Boolean(box && box.width > 0 && box.height > 0), 'native sheet drag has a real handle target');
    const x = box.x + box.width / 2, y = box.y + box.height / 2;
    await page.mouse.move(x, y);
    await page.mouse.down();
    await page.mouse.move(x, y + (expanded ? -120 : 120), { steps: 10 });
    await page.mouse.up();
    await page.waitForTimeout(300);
    check(await handle.getAttribute('aria-expanded') === String(expanded), 'actual native ' + (expanded ? 'up' : 'down') + ' drag selects the expected snap state');
  }
}

async function hitVisible(control) {
  return control.evaluate(element => {
    const rect = element.getBoundingClientRect();
    let left = Math.max(rect.left, 0), right = Math.min(rect.right, innerWidth);
    let top = Math.max(rect.top, 0), bottom = Math.min(rect.bottom, innerHeight);
    for (let node = element.parentElement; node; node = node.parentElement) {
      const style = getComputedStyle(node);
      if (/(auto|scroll|hidden|clip)/.test(style.overflowY + style.overflowX)) {
        const clip = node.getBoundingClientRect();
        left = Math.max(left, clip.left); right = Math.min(right, clip.right);
        top = Math.max(top, clip.top); bottom = Math.min(bottom, clip.bottom);
      }
    }
    const visible = right > left && bottom > top;
    const hit = visible ? document.elementFromPoint((left + right) / 2, (top + bottom) / 2) : null;
    return { visible, hit: Boolean(hit && (element === hit || element.contains(hit))), rect: rect.toJSON() };
  });
}

}

    check(applyHit.visible && applyHit.hit, 'existing filter apply action stays reachable inside its own scroll container', applyHit);

    await page.keyboard.press('Escape');
    await filterDialog.waitFor({ state: 'hidden' });
    check(await filter.evaluate(element => document.activeElement === element && !element.closest('[inert]')), 'filter close restores its own trigger focus and background interactivity');

  await scenario('nojs-long-normal-body-bot-metadata-and404', async () => {
    const page = await pageFor({ javaScriptEnabled: false });
    const variant = fixture.variants.long, path = '/incidents/' + variant.article_id;

    // page JavaScript is disabled, even for an already visible stable source.

      const readingBounds = await readingFor(dialog).boundingBox();
      await page.mouse.move(readingBounds.x + readingBounds.width / 2, readingBounds.y + readingBounds.height / 2);
      await page.mouse.wheel(0, 900);
      await page.waitForTimeout(200);
      sourceHit = await hitVisible(source);
      if (sourceHit.visible && sourceHit.hit) break;
    }
    await settled(dialog, 'nojs-after-native-scroll');
    check(sourceHit.visible && sourceHit.hit, 'NoJS original source is reachable through native reading scroll', sourceHit);
