async function auditIcons(page,label){
 const values=await page.evaluate(()=>{
  function textRects(text){
   const walker=document.createTreeWalker(text,NodeFilter.SHOW_TEXT),rects=[];let node;
   const box=text.getBoundingClientRect();
   while(node=walker.nextNode()){
    if(!node.textContent.trim()||node.parentElement.closest('.sr-only,[hidden]'))continue;
    const range=document.createRange();range.selectNodeContents(node);
    for(const r of range.getClientRects()){
     const left=Math.max(r.left,box.left),right=Math.min(r.right,box.right),top=Math.max(r.top,box.top),bottom=Math.min(r.bottom,box.bottom);
     if(right>left&&bottom>top)rects.push({left,right,top,bottom});
    }
   }
   if(!rects.length)return null;
   return {left:Math.min(...rects.map(r=>r.left)),right:Math.max(...rects.map(r=>r.right)),top:Math.min(...rects.map(r=>r.top)),bottom:Math.max(...rects.map(r=>r.bottom))};
  }
  return [...document.querySelectorAll('svg[class*="size-icon-"]')].filter(svg=>svg.getBoundingClientRect().width>0&&!svg.closest('[hidden]')).map(svg=>{
   const box=svg.getBoundingClientRect(),sizeName=[...svg.classList].find(c=>c.startsWith('size-icon-')),expected={indicator:8,compact:12,default:14,action:16}[sizeName?.replace('size-icon-','')];
   const wrap=svg.parentElement,text=[...wrap.children].find(el=>el!==svg&&el.tagName.toLowerCase()==='span'),r=text&&textRects(text);
   const columnGap=parseFloat(getComputedStyle(wrap).columnGap),position=text&&text.compareDocumentPosition(svg)&Node.DOCUMENT_POSITION_FOLLOWING?'end':'start';
   return {sizeName,expected,box:box.toJSON(),ariaHidden:svg.getAttribute('aria-hidden'),focusable:svg.getAttribute('focusable'),text:text?.textContent.trim().slice(0,130),textBox:r,gap:r?(position==='end'?box.left-r.right:r.left-box.right):null,cssGap:Number.isFinite(columnGap)?columnGap:null,centerDelta:r?Math.abs((box.top+box.bottom)/2-(r.top+r.bottom)/2):null};
  });
 });
 report.iconGeometry??=[];report.iconGeometry.push({label,width:await page.evaluate(()=>innerWidth),items:values});
 check(values.every(v=>v.expected&&Math.abs(v.box.width-v.expected)<.15&&Math.abs(v.box.height-v.expected)<.15&&v.ariaHidden==='true'&&v.focusable==='false'),label+' semantic SVG size and hidden decoration',{count:values.length,failures:values.filter(v=>!v.expected||Math.abs(v.box.width-v.expected)>=.15||Math.abs(v.box.height-v.expected)>=.15||v.ariaHidden!=='true'||v.focusable!=='false')});
 const paired=values.filter(v=>v.textBox);
 check(paired.every(v=>[4,6,8].includes(v.cssGap)&&Math.abs(v.gap-v.cssGap)<=1.1&&v.centerDelta<=1.5),label+' actual textRange gap and vertical center',{count:paired.length,failures:paired.filter(v=>![4,6,8].includes(v.cssGap)||Math.abs(v.gap-v.cssGap)>1.1||v.centerDelta>1.5)});
}
