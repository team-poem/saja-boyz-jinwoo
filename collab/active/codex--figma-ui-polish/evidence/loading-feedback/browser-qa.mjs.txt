function check(ok, name, evidence) {
  report.checks.push({ name, passed: Boolean(ok), ...(evidence === undefined ? {} : { evidence }) });
  return Boolean(ok);
}
function requireCheck(ok, name, evidence) {
  if (!check(ok, name, evidence)) throw new Error(name);
}
async function control(path, values = {}) {
  const url = new URL('/__qa/' + path, api);
  for (const [key, value] of Object.entries(values)) if (value !== undefined) url.searchParams.set(key, String(value));
  const response = await fetch(url);
  if (!response.ok) throw new Error('Fixture control failed: ' + response.status);
  return response.json();
}
async function state(label) {
  const value = await control('state');
  if (label) report.fixtureStates.push({ label, value });
  return value;
}
async function waitPending(scope, key, after = 0, timeout = 4000) {
  const started = Date.now();
  while (Date.now() - started < timeout) {
    const value = await state();
    const match = value.pending.find(item => item.scope === scope && (!key || item.key === key) && item.token > after);
    if (match) return match;
    await new Promise(resolve => setTimeout(resolve, 50));
  }
  throw new Error('Expected gated request did not arrive: ' + scope);
}

const browser = await chromium.launch({ channel: 'chrome', headless: true });
const contexts = new Set();
async function pageFor(options = {}) {
  const { blockPrefetch = false, ...contextOptions } = options;
  const context = await browser.newContext({
    viewport: { width: 393, height: 852 },
    reducedMotion: 'reduce',
    permissions: ['geolocation'],
    geolocation: { latitude: 37.56, longitude: 126.99, accuracy: 1000 },
    ...contextOptions,
  });
  contexts.add(context);
  // Only RSC loader/retry contexts intercept prefetch. Cache/NoJS/SDK-image
  // verification contexts have no routing interception at all.
  if (blockPrefetch) await context.route('**/*', async route => {
    const headers = route.request().headers();
    if (headers['next-router-prefetch'] || headers.purpose === 'prefetch') return route.abort();
    return route.continue();
  });
  const page = await context.newPage();
  page.on('pageerror', error => report.pageErrors.push({ message: error.message }));
  page.on('request', request => {
    if (!['GET', 'HEAD', 'OPTIONS'].includes(request.method())) {
      // Observational guard only: no steps in this runner target a write action.
      report.observedWrites.push({ method: request.method(), origin: new URL(request.url()).origin, path: new URL(request.url()).pathname });
    }
  });
  page.on('response', response => {
    const url = new URL(response.url());
    if (url.hostname === 'nrbe.map.naver.net' && response.status() === 200) report.tiles200++;
  });
  return page;
}
async function closePage(page) {
  const context = page.context();
  contexts.delete(context);
  await context.close();
}
async function scenario(name, run) {
  const before = report.checks.length;
  const entry = { name, started: Date.now(), completed: false };
  report.scenarios.push(entry);
  try { await control('reset'); await run(); entry.completed = true; }
  catch (error) { entry.error = String(error?.stack || error); check(false, name + ' completed', entry.error); }
  finally {
    await state(name + ':end').catch(() => {});
    await control('reset').catch(() => {});
    for (const context of [...contexts]) { contexts.delete(context); await context.close().catch(() => {}); }
    entry.elapsedMs = Date.now() - entry.started;
    entry.checks = report.checks.length - before;
  }
}

async function loadedImage(figure) {
  await figure.locator('img').waitFor();

  await figure.locator(skeletonSelector).waitFor({ state: 'detached' });
}
async function imageGeometry(figure) {
  return figure.evaluate(element => {
    const image = element.querySelector('img'), loading = element.querySelector('[data-image-loading]');
    return {
      figure: element.getBoundingClientRect().toJSON(),
      image: image?.getBoundingClientRect().toJSON(),
      skeleton: loading?.getBoundingClientRect().toJSON(),
      skeletonHidden: loading?.getAttribute('aria-hidden'),
      complete: image?.complete, naturalWidth: image?.naturalWidth,
      caption: element.querySelector('figcaption')?.textContent,
    };
  });
}

    await control('release', { scope: 'detail', key: id, count: 1, status: 200 });
    const response = await navigation;
    check(response.status() === 200, 'delayed direct document returns200');
    await page.getByRole('heading', { name: title, exact: true }).waitFor();
    check((await state()).requests.filter(item => item.scope === 'detail' && item.key === id).length === 2, 'direct probe and cached page/metadata load remain two upstream GETs');
    report.directTiming = { elapsedMs: Date.now() - started, probeToken: probe.token, detailToken: detail.token, preHtmlDelayIsExpected: true };
    await snapshot(page, 'direct-delayed-success', 'Both direct-request stages were gated and explicitly released; initial pre-HTML delay is expected.');
  });
  await scenario('nojs-visible-body-bot-metadata-and-real404', async () => {
    const page = await pageFor({ javaScriptEnabled: false });
    const response = await page.goto(origin + detailPath, { waitUntil: 'load' });
    check(response.status() === 200, 'NoJS valid article document status200');
    await page.getByRole('heading', { name: title, exact: true }).waitFor({ state: 'visible' });
    check(await page.getByRole('region', { name: '기사 요약' }).getByText(summary, { exact: true }).isVisible(), 'NoJS actual article summary is visible');
    check(await page.getByRole('link', { name: /기사 원문/ }).isVisible(), 'NoJS actual source link is visible');
    const image = page.locator(imageSelector);
    const imageState = await image.evaluate(image => ({ complete: image.complete, naturalWidth: image.naturalWidth, width: image.getBoundingClientRect().width, height: image.getBoundingClientRect().height }));
    check(imageState.complete && imageState.naturalWidth > 0 && imageState.width > 0 && imageState.height > 0, 'NoJS image loads visibly without a client onLoad callback', imageState);
    await snapshot(page, 'direct-nojs-valid', 'JavaScript disabled; final article title, summary, source and image must remain visible.');
    const bot = await fetch(origin + detailPath, { headers: { 'user-agent': 'Googlebot' } });
    const html = await bot.text(), doc = new JSDOM(html).window.document;
    check(bot.status === 200, 'native fetch bot response status is numeric200');
    check(doc.querySelector('title')?.textContent.includes(title), 'bot receives the actual article title');
    check(doc.querySelector('link[rel="canonical"]')?.getAttribute('href') === siteOrigin + detailPath, 'bot canonical is the stable public article URL');
    check(!/noindex/.test(doc.querySelector('meta[name="robots"]')?.getAttribute('content') || ''), 'successful bot article is indexable');
    const missingResponse = await page.goto(origin + '/incidents/' + missing + '?qa=loading', { waitUntil: 'load' });
    check(missingResponse.status() === 404, 'NoJS missing article keeps a real404');
    check(/noindex/.test(await page.locator('meta[name="robots"]').getAttribute('content')), 'NoJS missing article is noindex');
    check(await page.getByRole('heading', { level: 1 }).isVisible(), 'NoJS missing article has a visible explanation');
    check(await page.getByRole('link', { name: /목록/ }).first().isVisible() && await page.getByRole('link', { name: /지도/ }).first().isVisible(), 'NoJS404 retains visible list/map escape paths');
    check(new URL(page.url()).pathname === '/incidents/' + missing && new URL(page.url()).searchParams.get('qa') === 'loading', 'NoJS404 preserves the original URL and query');
    await snapshot(page, 'direct-nojs404', 'Actual upstream404, no JavaScript, unchanged original address and visible escape paths.');
  });
} finally {
  await state('final').catch(() => {});
  await control('reset').catch(() => {});
  for (const context of contexts) await context.close().catch(() => {});
  await browser.close();
  check(report.observedWrites.filter(request => request.origin === origin).length === 0, 'runner initiated no application write requests', report.observedWrites);
  report.providerPosts = report.observedWrites.filter(request => request.origin !== origin);
  check(report.pageErrors.length === 0, 'no uncaught browser page errors', report.pageErrors);
  check(report.tiles200 > 0, 'actual Naver raster tiles loaded without SDK stub', { tiles200: report.tiles200 });
  report.finished = new Date().toISOString();
  report.summary = {
    checks: report.checks.length,
    failed: report.checks.filter(item => !item.passed).length,
    scenarios: report.scenarios.length,
    incompleteScenarios: report.scenarios.filter(item => !item.completed).length,
    screenshots: report.screenshots.length,
    pageErrors: report.pageErrors.length,
  };
  await writeFile(output + '/report.json', JSON.stringify(report, null, 2));
  await writeFile(output + '/manifest.json', JSON.stringify({
    sourceHead: report.sourceHead, buildId, synthetic: true, personalLocation: false,
    SDKStubbed: false, SDKModified: false, summary: report.summary,
    screenshots: report.screenshots, report: 'report.json',
    execution: 'Run only after parent QA-ready; this script does not build or start any server.',
  }, null, 2));
}
console.log(JSON.stringify({ buildId, output, ...report.summary }));
