 const top=sticky?.getBoundingClientRect().bottom??0,nav=document.querySelector('nav[aria-label="주 메뉴"]'),bottom=nav?.getBoundingClientRect().top??innerHeight;
 const first=parts[0],hit=first?document.elementFromPoint((first.left+first.right)/2,(first.top+first.bottom)/2):null;
 return {text:a.textContent.trim(),href:new URL(a.href).pathname,unclippedBox:a.getBoundingClientRect().toJSON(),paintedParts:parts,top,bottom,hitSelf:hit===a||a.contains(hit),fullyVisible:parts.length>0&&parts.every(r=>r.top>=top-1&&r.bottom<=bottom+1),scrollY};
 });}
